% ── Graceful missing-figure fallback ────────────────────────────────────────
% Until the team drops the sourced .jpg/.png files into figures/, any missing
% image is replaced by a labelled placeholder box so the document still
% compiles. Once the real file exists, it is used automatically.
\usepackage{xparse}
\let\origincludegraphics\includegraphics
\RenewDocumentCommand{\includegraphics}{O{}m}{%
  \IfFileExists{#2}{\origincludegraphics[#1]{#2}}{%
    \fbox{\parbox[c][3.2cm][c]{0.78\linewidth}{\centering\ttfamily\small
      [ missing figure ]\\[2pt]#2\\[2pt]\textnormal{\itshape see FIGURE\_GUIDE.md}}}%
  }%
}
